3. What is the method used to iterate over the "values" of a dictionary?
some_list = [0, 7, 12]
some_dict = {"key1": "value1", "key2": 2, "key3": "value3"}

for key in some_dict:
    print(key) # key1...key2...key3
    print(some_dict[key])

for value in some_dict.values():
    print(value)

for key, value in some_dict.items():
    print(key)
    print(value)
